action_demand_subjugation = {
	consider = {
		CR_CONSIDER = {
			weight = { weight = 50 }
			trigger = {
				has_ai_personality = cunning_raiders
			}
		}
		MB_CONSIDER = {
			weight = { weight = 50 }
			trigger = {
				has_ai_personality = mirthful_bandits
			}
		}
	}
	select = {
		# Vassalization Select
		 CR_DEMAND_VASS_SELECT_1 = {
			weight = { weight = 50 }
			trigger = {
				has_ai_personality = cunning_raiders
				OR = {
					is_friendly_to = FROM
					is_cordial_to = FROM
					is_protective_to = FROM
				}
			}
		}
		# Vassalization Select (Hostile)
		CR_DEMAND_VASS_SELECT_2 = {
			weight = { weight = 50 }
			trigger = {
				has_ai_personality = cunning_raiders
				NOT = {
					is_friendly_to = FROM
					is_cordial_to = FROM
					is_protective_to = FROM
				}
			}
		}
	}
	propose = {
		CR_DEMAND_VASS_PROPOSE = {
			weight = { weight = 50 }
			trigger = {
				from = { has_ai_personality = cunning_raiders }
			}
		}
	}
	accept = {
		# Accept
		CR_DEMAND_VASS_ACCEPT_1 = {
			weight = { weight = 50 }
			trigger = {
				has_ai_personality = cunning_raiders
			}
		}
		# Accept Mirthful Bandits
		CR_DEMAND_VASS_ACCEPT_2 = {
			trigger = {
				has_ai_personality = cunning_raiders
				FROM = { has_ai_personality = mirthful_bandits }
			}
		}
		# Accept other pirates
		CR_DEMAND_VASS_ACCEPT_3 = {
			trigger = {
				has_ai_personality = cunning_raiders
				FROM = { is_pirate_empire = yes }
			}
		}
	}
	refuse = {
		# Refuse
		CR_DEMAND_VASS_REFUSE_1 = {
			weight = { weight = 50 }
			trigger = {
				has_ai_personality = cunning_raiders
				OR = {
					is_friendly_to = FROM
					is_cordial_to = FROM
					is_protective_to = FROM
				}
			}
		}
		# Refuse Hostile
		CR_DEMAND_VASS_REFUSE_2 = {
			weight = { weight = 50 }
			trigger = {
				has_ai_personality = cunning_raiders
				NOT = {
					is_friendly_to = FROM
					is_cordial_to = FROM
					is_protective_to = FROM
				}
			}
		}
	}
}
