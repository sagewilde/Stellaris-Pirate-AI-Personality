### Vanilla Overwrites

values_others = {
	AND = {
		OR = {
			is_xenophile = yes
			has_valid_civic = civic_hive_empath
			has_valid_civic = civic_machine_servitor
			has_valid_civic = civic_machine_exploration_protocol
			has_origin = origin_fruitful
			has_animal_handler_job = yes
			has_symbiosis_integrator_job = yes
			has_valid_civic = civic_hive_familiar_face
			has_hippie_government = yes					# For compatibility with Hippie AI Personalities
			has_ai_personality = mirthful_bandits
		}
		NOT = {
			has_menace_perk = menp_behemoth_ever_hungry
		}
	}
}

# Pirates

is_pirate_empire = {
    OR = {
        has_valid_civic = civic_crusader_spirit_corporate
		has_valid_civic = civic_void_reavers
        has_technology = tech_pirate_leaders
    }
}

is_impoverished = {
	is_country_type = default
	OR = {
		AND = {
			NOR = {
				is_species_class = ROBOT
				is_species_class = MACHINE
				is_species_class = LITHOID
			}
			exists = owner
			owner = { 
				has_country_resource = { 
					type = food 
					amount < 100 
				}
				has_monthly_income = {
					resource = food
					value < 0
				}
			}
		}
		AND = {
			is_species_class = LITHOID
			exists = owner
			owner = { 
				has_country_resource = { 
					type = minerals 
					amount < 100 
				}
				has_monthly_income = {
					resource = minerals
					value < 0
				}
			}
		}
		AND = {
			OR = {
				is_species_class = ROBOT
				is_species_class = MACHINE
			}
			exists = owner
			owner = {
				has_country_resource = { 
					type = energy
					amount < 100 
				}
				has_monthly_income = {
					resource = energy
					value < 0
				}
			}
		}
	}
}

is_weak_and_rich = {
	AND = {
		relative_power = {
			who = root
			category = economy
			value >= equivalent
		}
		relative_power = {
			who = root
			category = military
			value <= equivalent
		}
	}
}

is_special_personality = {
	OR = {
		has_ai_personality = "peaceful_counterculture"
		has_ai_personality = "mirthful_bandits"
		has_ai_personality = "cunning_raiders"
	}
}
