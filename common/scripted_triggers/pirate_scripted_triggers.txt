			is_xenophile = yes
			has_valid_civic = civic_hive_empath
			has_valid_civic = civic_machine_servitor
			has_valid_civic = civic_machine_exploration_protocol
			has_origin = origin_fruitful
			has_animal_handler_job = yes
			has_symbiosis_integrator_job = yes
			has_valid_civic = civic_hive_familiar_face
			#has_hippie_government = yes
			has_ai_personality = mirthful_bandits
		}
		NOT = {
			has_menace_perk = menp_behemoth_ever_hungry
		}
	}
}

# Pirates

is_pirate_empire = {
    OR = {
        has_valid_civic = civic_crusader_spirit_corporate
		has_valid_civic = civic_void_reavers
        has_technology = tech_pirate_leadership
    }
}

is_impoverished = {
	OR = {
		AND = {
			is_lithoid_empire = no
			is_robotic_species = no
			has_country_resource = { 
				type = food 
				amount < 450 
			}
			has_monthly_income = {
				resource = food
				value < 1
			}
		}
		AND = {
			is_lithoid_empire = yes
			has_country_resource = { 
				type = minerals 
				amount < 450 
			}
			has_monthly_income = {
				resource = minerals
				value < 1
			}
		}
		AND = {
			is_robotic_species = yes
			has_country_resource = { 
				type = energy
				amount < 450 
			}
			has_monthly_income = {
				resource = energy
				value < 1
			}
		}
	}
}

is_weak_and_rich = {
	AND = {
		relative_power = {
			who = root
			category = economy
			value >= equivalent
		}
		relative_power = {
			who = root
			category = military
			value <= equivalent
		}
	}
}
