# Pirates

has_not_distar = {
	NOT = {	host_has_dlc = "Distant Stars Story Pack" }
}

is_impoverished = {
	is_country_type = default
	OR = {
		AND = {
			NOR = {
				is_species_class = ROBOT
				is_species_class = MACHINE
				is_species_class = LITHOID
			}
			exists = owner
			owner = { 
				has_country_resource = { 
					type = food 
					amount < 100 
				}
				has_monthly_income = {
					resource = food
					value < 0
				}
			}
		}
		AND = {
			is_species_class = LITHOID
			exists = owner
			owner = { 
				has_country_resource = { 
					type = minerals 
					amount < 100 
				}
				has_monthly_income = {
					resource = minerals
					value < 0
				}
			}
		}
		AND = {
			OR = {
				is_species_class = ROBOT
				is_species_class = MACHINE
			}
			exists = owner
			owner = {
				has_country_resource = { 
					type = energy
					amount < 100 
				}
				has_monthly_income = {
					resource = energy
					value < 0
				}
			}
		}
	}
}
